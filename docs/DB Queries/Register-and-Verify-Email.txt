1. Find user by email (with role JOIN via @EntityGraph)
SELECT u.*, r.*
FROM app_user u
LEFT JOIN role r ON u.role_id = r.role_id
WHERE u.email = ?

2. Find user by phone (with role JOIN via @EntityGraph)
SELECT u.*, r.*
FROM app_user u
LEFT JOIN role r ON u.role_id = r.role_id
WHERE u.phone = ?

3. Find role by name
SELECT *
FROM role
WHERE role_name = 'Customer'
